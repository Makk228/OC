#!/bin/bash

CONFIG_FILE="./watcher.conf"

if [[ ! -f "$CONFIG_FILE" ]]; then
    echo "Config file not found: $CONFIG_FILE"
    exit 1
fi

source "$CONFIG_FILE"

find . -maxdepth 1 -type f -name "$(basename "$LOG").20*" -mtime +7 -delete
find "$REPORT_DIR" -type f -name "report_*.txt" -mtime +14 -delete
