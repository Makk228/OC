#!/bin/bash

CONFIG_FILE="./watcher.conf"

if [[ ! -f "$CONFIG_FILE" ]]; then
    cat > "$CONFIG_FILE" <<'EOF'
PIPE="./watcher.fifo"
LOG="./watcher.log"
PIDFILE="./watcher.pid"
SLEEP_INTERVAL=2
LOG_LEVEL="INFO"
REPORT_DIR="./reports"
EOF
fi

source "$CONFIG_FILE"

running=true
reload_requested=false
status_requested=false
rotate_requested=false
event_count=0
start_time=$(date +%s)
fifo_opened=false

level_to_num() {
    case "$1" in
        DEBUG) echo 0 ;;
        INFO)  echo 1 ;;
        WARN)  echo 2 ;;
        ERROR) echo 3 ;;
        *)     echo 1 ;;
    esac
}

should_log() {
    local msg_level="$1"
    local current_level
    local msg_num
    local current_num

    current_level="${LOG_LEVEL:-INFO}"
    msg_num=$(level_to_num "$msg_level")
    current_num=$(level_to_num "$current_level")

    [[ "$msg_num" -ge "$current_num" ]]
}

log_message() {
    local level="$1"
    local message="$2"

    if should_log "$level"; then
        echo "$(date '+%Y-%m-%d %H:%M:%S') [$level] $message" >> "$LOG"
    fi
}

init_paths() {
    mkdir -p "$REPORT_DIR"
}

init_fifo() {
    if [[ ! -p "$PIPE" ]]; then
        rm -f "$PIPE"
        mkfifo "$PIPE"
    fi
}

open_fifo() {
    init_fifo
    exec 3<> "$PIPE"
    fifo_opened=true
}

close_fifo() {
    if "$fifo_opened"; then
        exec 3>&-
        exec 3<&-
        fifo_opened=false
    fi
}

write_pid() {
    echo $$ > "$PIDFILE"
}

cleanup() {
    close_fifo
    rm -f "$PIDFILE"
}

end_function() {
    log_message "INFO" "Watcher stopping"
    cleanup
    exit 0
}

request_reload() {
    reload_requested=true
}

request_status() {
    status_requested=true
}

request_rotate() {
    rotate_requested=true
}

do_reload() {
    close_fifo
    source "$CONFIG_FILE"
    init_paths
    open_fifo
    log_message "INFO" "Configuration reloaded"
}

do_status() {
    local now uptime
    now=$(date +%s)
    uptime=$((now - start_time))

    log_message "INFO" "STATUS pid=$$ events=$event_count uptime=${uptime}s pipe=$PIPE log=$LOG interval=$SLEEP_INTERVAL level=$LOG_LEVEL"
}

do_rotate() {
    local ts archived_log
    ts=$(date '+%Y%m%d_%H%M%S')
    archived_log="${LOG}.${ts}"

    if [[ -f "$LOG" ]]; then
        mv "$LOG" "$archived_log"
    fi

    touch "$LOG"
    log_message "INFO" "Log rotated, previous file: $archived_log"
}

handle_flag() {
    local flag_name="$1"
    local action="$2"
    declare -n flag_ref="$flag_name"

    if "$flag_ref"; then
        "$action"
        flag_ref=false
    fi
}

process_fifo_message() {
    local line="$1"

    if [[ -z "$line" ]]; then
        log_message "WARN" "Empty FIFO message ignored"
        return 0
    fi

    case "$line" in
        STATUS)
            log_message "INFO" "FIFO command STATUS received"
            status_requested=true
            ;;
        STOP)
            log_message "INFO" "FIFO command STOP received"
            end_function
            ;;
        RELOAD)
            log_message "INFO" "FIFO command RELOAD received"
            reload_requested=true
            ;;
        ROTATE)
            log_message "INFO" "FIFO command ROTATE received"
            rotate_requested=true
            ;;
        MODE_CHANGE)
            log_message "INFO" "FIFO command MODE_CHANGE received"
            ;;
        *)
            ((event_count++))
            log_message "INFO" "FIFO event: $line"
            ;;
    esac
}

main_loop() {
    init_paths
    write_pid
    open_fifo
    log_message "INFO" "Watcher started"

    while "$running"; do
        handle_flag reload_requested do_reload
        handle_flag status_requested do_status
        handle_flag rotate_requested do_rotate

        if IFS= read -r -t 1 line <&3; then
            process_fifo_message "$line"
        fi

        sleep "$SLEEP_INTERVAL"
    done
}

is_running() {
    if [[ -f "$PIDFILE" ]]; then
        local pid
        pid=$(cat "$PIDFILE")
        kill -0 "$pid" 2>/dev/null
        return $?
    fi
    return 1
}

start_daemon() {
    if is_running; then
        echo "Watcher already running with PID $(cat "$PIDFILE")"
        exit 1
    fi

    nohup "$0" run >/dev/null 2>&1 &
    sleep 1

    if is_running; then
        echo "Watcher started with PID $(cat "$PIDFILE")"
    else
        echo "Failed to start watcher"
        exit 1
    fi
}

stop_daemon() {
    if is_running; then
        local pid
        pid=$(cat "$PIDFILE")
        kill -TERM "$pid"
        echo "Watcher stop signal sent to PID $pid"
    else
        echo "Watcher is not running"
        rm -f "$PIDFILE"
    fi
}

status_daemon() {
    if is_running; then
        echo "Watcher is running with PID $(cat "$PIDFILE")"
    else
        echo "Watcher is not running"
    fi
}

trap end_function TERM INT
trap request_reload HUP
trap request_status USR1
trap request_rotate USR2

case "$1" in
    start)
        start_daemon
        ;;
    stop)
        stop_daemon
        ;;
    status)
        status_daemon
        ;;
    run)
        main_loop
        ;;
    *)
        echo "Usage: $0 {start|stop|status|run}"
        exit 1
        ;;
esac
