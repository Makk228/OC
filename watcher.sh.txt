#!/bin/bash

CONFIG_FILE="./watcher.conf"

if [[ ! -f "$CONFIG_FILE" ]]; then
    cat > "$CONFIG_FILE" <<'EOF'
PIPE="./watcher.fifo"
LOG="./watcher.log"
PIDFILE="./watcher.pid"
SLEEP_INTERVAL=2
LOG_LEVEL="INFO"
EOF
fi

source "$CONFIG_FILE"

running=true
reload_requested=false
status_requested=false
rotate_requested=false
event_count=0
start_time=$(date +%s)

log_message() {
    local level="$1"
    local message="$2"
    echo "$(date '+%Y-%m-%d %H:%M:%S') [$level] $message" >> "$LOG"
}

init_fifo() {
    if [[ ! -p "$PIPE" ]]; then
        rm -f "$PIPE"
        mkfifo "$PIPE"
    fi
}

write_pid() {
    echo $$ > "$PIDFILE"
}

cleanup() {
    rm -f "$PIDFILE"
}

end_function() {
    log_message "INFO" "Watcher stopping"
    exec 3>&-
    exec 3<&-
    cleanup
    exit 0
}

request_reload() {
    reload_requested=true
}

request_status() {
    status_requested=true
}

request_rotate() {
    rotate_requested=true
}

do_reload() {
    source "$CONFIG_FILE"
    log_message "INFO" "Configuration reloaded"
}

do_status() {
    local now uptime
    now=$(date +%s)
    uptime=$((now - start_time))
    log_message "INFO" "STATUS pid=$$ events=$event_count uptime=${uptime}s pipe=$PIPE log=$LOG interval=$SLEEP_INTERVAL"
}

do_rotate() {
    local ts new_log
    ts=$(date '+%Y%m%d_%H%M%S')
    new_log="${LOG}.${ts}"

    if [[ -f "$LOG" ]]; then
        mv "$LOG" "$new_log"
    fi

    touch "$LOG"
    log_message "INFO" "Log rotated, previous file: $new_log"
}

handle_flag() {
    local flag_name="$1"
    local action="$2"
    declare -n flag_ref="$flag_name"

    if "$flag_ref"; then
        "$action"
        flag_ref=false
    fi
}

process_fifo_message() {
    local line="$1"

    if [[ -z "$line" ]]; then
        return 0
    fi

    case "$line" in
        STATUS)
            status_requested=true
            ;;
        STOP)
            log_message "INFO" "FIFO command STOP received"
            end_function
            ;;
        MODE_CHANGE)
            log_message "INFO" "FIFO command MODE_CHANGE received"
            ;;
        *)
            ((event_count++))
            log_message "INFO" "FIFO event: $line"
            ;;
    esac
}

main_loop() {
    log_message "INFO" "Watcher started"
    write_pid
    init_fifo
    exec 3<> "$PIPE"

    while "$running"; do
        handle_flag reload_requested do_reload
        handle_flag status_requested do_status
        handle_flag rotate_requested do_rotate

        if IFS= read -r -t 1 line <&3; then
            process_fifo_message "$line"
        fi

        sleep "$SLEEP_INTERVAL"
    done
}

is_running() {
    if [[ -f "$PIDFILE" ]]; then
        local pid
        pid=$(cat "$PIDFILE")
        kill -0 "$pid" 2>/dev/null
        return $?
    fi
    return 1
}

start_daemon() {
    if is_running; then
        echo "Watcher already running with PID $(cat "$PIDFILE")"
        exit 1
    fi

    nohup "$0" run >/dev/null 2>&1 &
    echo "Watcher started"
}

stop_daemon() {
    if is_running; then
        local pid
        pid=$(cat "$PIDFILE")
        kill -TERM "$pid"
        echo "Watcher stop signal sent to PID $pid"
    else
        echo "Watcher is not running"
        rm -f "$PIDFILE"
    fi
}

status_daemon() {
    if is_running; then
        echo "Watcher is running with PID $(cat "$PIDFILE")"
    else
        echo "Watcher is not running"
    fi
}

trap end_function TERM INT
trap request_reload HUP
trap request_status USR1
trap request_rotate USR2

case "$1" in
    start)
        start_daemon
        ;;
    stop)
        stop_daemon
        ;;
    status)
        status_daemon
        ;;
    run)
        main_loop
        ;;
    *)
        echo "Usage: $0 {start|stop|status|run}"
        exit 1
        ;;
esac
