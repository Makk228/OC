#!/bin/bash

log_file="monitor_two.log"
: > "$log_file"

echo "time,total_mem,used_mem,free_mem,buff_cache,total_swap,used_swap,free_swap,pid,vsz,rss,cpu,mem,command" >> "$log_file"

while true
do
    timestamp=$(date '+%H:%M:%S')

    mem_line=$(free -m | awk '/^Mem:/ {print $2","$3","$4","$6}')
    swap_line=$(free -m | awk '/^Swap:/ {print $2","$3","$4}')

    ps -C bash -o pid=,rss= --sort=-rss | head -n 2 | while read pid rss
    do
        proc_line=$(ps -p "$pid" -o pid=,vsz=,rss=,%cpu=,%mem=,comm= 2>/dev/null | awk '{print $1","$2","$3","$4","$5","$6}')
        if [ -n "$proc_line" ]; then
            echo "$timestamp,$mem_line,$swap_line,$proc_line" >> "$log_file"
        fi
    done

    sleep 1
done
