#!/bin/bash

cd "$(dirname "$0")"

log_file="monitor_two.log"
: > "$log_file"

echo "time,total_mem,used_mem,free_mem,buff_cache,total_swap,used_swap,free_swap,process,pid,vsz,rss,cpu,mem,command" >> "$log_file"

while true
do
    timestamp=$(date '+%H:%M:%S')

    mem_line=$(free -m | awk '/^Mem:/ {print $2","$3","$4","$6}')
    swap_line=$(free -m | awk '/^Swap:/ {print $2","$3","$4}')

    pid1=$(cat mem1.pid 2>/dev/null)
    pid2=$(cat mem2.pid 2>/dev/null)

    if [ -n "$pid1" ]; then
        proc_line=$(ps -p "$pid1" -o pid=,vsz=,rss=,%cpu=,%mem=,comm= 2>/dev/null | awk '{print $1","$2","$3","$4","$5","$6}')
        if [ -n "$proc_line" ]; then
            echo "$timestamp,$mem_line,$swap_line,mem1,$proc_line" >> "$log_file"
        fi
    fi

    if [ -n "$pid2" ]; then
        proc_line=$(ps -p "$pid2" -o pid=,vsz=,rss=,%cpu=,%mem=,comm= 2>/dev/null | awk '{print $1","$2","$3","$4","$5","$6}')
        if [ -n "$proc_line" ]; then
            echo "$timestamp,$mem_line,$swap_line,mem2,$proc_line" >> "$log_file"
        fi
    fi

    sleep 1
done
