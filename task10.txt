#!/bin/bash

input_file="task9_result.txt"
output_file="task10_result.txt"

awk -F'[:= ]+' '
{
    pid = $2
    ppid = $5
    art = $8

    if (NR == 1) {
        current_ppid = ppid
        sum = 0
        count = 0
    }

    if (ppid != current_ppid) {
        avg = (count > 0) ? sum / count : 0
        printf "Average_Running_Children_of_ParentID=%s is %.6f\n", current_ppid, avg
        current_ppid = ppid
        sum = 0
        count = 0
    }

    print
    sum += art
    count++
}
END {
    if (count > 0) {
        avg = sum / count
        printf "Average_Running_Children_of_ParentID=%s is %.6f\n", current_ppid, avg
    }
}
' "$input_file" > "$output_file"
